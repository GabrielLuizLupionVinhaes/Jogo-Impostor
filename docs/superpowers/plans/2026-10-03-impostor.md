**Goal:** Construir localmente o protótipo jogável do "Jogo do Impostor": um site estático, feito primeiro para celular e pronto para o GitHub Pages.

**Architecture:**
- HTML, CSS e JavaScript puro com ES modules, sem build e sem dependências.
- As regras ficam em `js/game.js`, um módulo puro testado com `node --test`. O banco de palavras fica em `js/words.js` e a persistência em `js/storage.js`.
- A interface fica em `js/app.js`, que troca as telas de uma página única.
- Um servidor Node sem dependências permite testar no PC e no celular.

**Tech Stack:** HTML5, CSS3, JavaScript ES2020 (ES modules), Node.js 24 (`node:test`, `node:http`), git. Para verificar no navegador, `playwright-core` instalado só no scratchpad, fora do projeto, dirigindo o Chrome que já está instalado.

## Contexto

O usuário quer um jogo de festa no estilo "Impostor", para jogar presencialmente passando um celular de mão em mão.

- Todos recebem a mesma palavra secreta, menos o impostor, que recebe só uma palavra-dica.
- O app sorteia quem começa a falar e, no fim, revela o impostor.

O design foi aprovado no brainstorming e está em `docs/superpowers/specs/2026-10-03-impostor-design.md`, já commitado. O repositório git local tem só essa spec. Tudo roda **localmente agora**; o deploy no GitHub Pages fica para depois.

A revisão técnica do design acrescentou alguns refinamentos, registrados na spec no Task 1:

- esconder a palavra quando o app vai para segundo plano;
- proteção contra toque duplo;
- tamanho da palavra calculado pelo comprimento;
- bloqueio do "puxar para atualizar";
- servidor com checagem de maiúsculas e minúsculas.

```js
// Regras puras do Jogo do Impostor: limites, nomes, configurações e sorteios.
// Nada aqui acessa a tela ou o armazenamento, então tudo é testável com `node --test`.

export const MIN_PLAYERS = 3;
export const MAX_PLAYERS = 20;
export const MAX_NAME_LENGTH = 20;
export const DEFAULT_PLAYER_COUNT = 4;

/** Impostores precisam ser minoria: no máximo floor((n - 1) / 2), e pelo menos 1. */
export function maxImpostors(playerCount) {
  return Math.max(1, Math.floor((playerCount - 1) / 2));
}

/** 3–6 jogadores → 1; 7–12 → 2; 13 ou mais → 3 (sem passar do máximo). */
export function recommendedImpostors(playerCount) {
  const recommended = playerCount <= 6 ? 1 : playerCount <= 12 ? 2 : 3;
  return Math.min(recommended, maxImpostors(playerCount));
}

export function clampImpostors(count, playerCount) {
  return Math.min(Math.max(count, 1), maxImpostors(playerCount));
}

/** Texto para comparação: sem acentos, minúsculo e sem espaços nas pontas. */
export function normalizeText(text) {
  return String(text).normalize('NFD').replace(/[\u0300-\u036f]/g, '').toLowerCase().trim();
}

export function defaultName(position) {
  return `Jogador ${position}`;
}

/** Próximo "Jogador N" que ainda não existe na lista. */
export function nextDefaultName(names) {
  const taken = new Set(names.map(normalizeText));
  let position = names.length + 1;
  while (taken.has(normalizeText(defaultName(position)))) position += 1;
  return defaultName(position);
}

/** Nomes prontos para a rodada: sem espaços nas pontas, até 20 caracteres, vazio → "Jogador {posição}". */
export function normalizePlayers(names) {
  return names.map((name, index) => {
    const clean = String(name ?? '').trim().slice(0, MAX_NAME_LENGTH).trim();
    return clean || defaultName(index + 1);
  });
}

/** Nomes repetidos (sem diferenciar maiúsculas e acentos), cada um uma vez, na grafia da primeira ocorrência. */
export function findDuplicateNames(names) {
  const firstSeen = new Map();
  const duplicates = [];
  for (const name of names) {
    const key = normalizeText(name);
    if (!key) continue;
    if (!firstSeen.has(key)) {
      firstSeen.set(key, String(name).trim());
    } else if (!duplicates.includes(firstSeen.get(key))) {
      duplicates.push(firstSeen.get(key));
    }
  }
  return duplicates;
}

export function defaultSettings(allThemeIds) {
  return {
    players: Array.from({ length: DEFAULT_PLAYER_COUNT }, (_, index) => defaultName(index + 1)),
    impostors: 1,
    themeIds: [...allThemeIds],
    hints: true,
  };
}

/** Corrige configurações salvas (inválidas, antigas ou de outra versão). */
export function sanitizeSettings(raw, allThemeIds) {
  if (!raw || typeof raw !== 'object' || Array.isArray(raw)) return defaultSettings(allThemeIds);

  const players = Array.isArray(raw.players)
    ? raw.players.filter((name) => typeof name === 'string').map((name) => name.slice(0, MAX_NAME_LENGTH))
    : defaultSettings(allThemeIds).players;
  players.splice(MAX_PLAYERS);
  while (players.length < MIN_PLAYERS) players.push(nextDefaultName(players));

  const savedThemes = Array.isArray(raw.themeIds) ? raw.themeIds : [];
  let themeIds = allThemeIds.filter((id) => savedThemes.includes(id));
  if (themeIds.length === 0) themeIds = [...allThemeIds];

  const impostors = Number.isInteger(raw.impostors)
    ? clampImpostors(raw.impostors, players.length)
    : recommendedImpostors(players.length);

  return {
    players,
    impostors,
    themeIds,
    hints: typeof raw.hints === 'boolean' ? raw.hints : true,
  };
}
```

```js
// Gerador pseudoaleatório com semente (mulberry32): os mesmos sorteios a cada execução dos testes.
export function seededRng(seed) {
  let state = seed >>> 0;
  return () => {
    state = (state + 0x6d2b79f5) >>> 0;
    let t = state;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
```

```js
/** Inteiro aleatório em [0, max). */
function randomInt(max, rng) {
  return Math.min(max - 1, Math.floor(rng() * max));
}

/** Cópia embaralhada (Fisher–Yates); não altera a lista original. */
export function shuffle(items, rng = Math.random) {
  const copy = [...items];
  for (let i = copy.length - 1; i > 0; i -= 1) {
    const j = randomInt(i + 1, rng);
    [copy[i], copy[j]] = [copy[j], copy[i]];
  }
  return copy;
}

export function wordId(themeId, word) {
  return `${themeId}/${word}`;
}

/**
 * Sorteia um tema entre os marcados e uma palavra ainda não usada desse tema.
 * Quando o tema se esgota, só as palavras dele voltam a ficar disponíveis.
 */
export function pickWord(themes, selectedThemeIds, usedWordIds, rng = Math.random) {
  const selected = themes.filter((theme) => selectedThemeIds.includes(theme.id));
  if (selected.length === 0) throw new Error('Nenhum tema selecionado');

  const theme = selected[randomInt(selected.length, rng)];
  const themeWordIds = new Set(theme.words.map((entry) => wordId(theme.id, entry.word)));
  const used = new Set(usedWordIds);
  let available = theme.words.filter((entry) => !used.has(wordId(theme.id, entry.word)));
  let nextUsed = [...usedWordIds];
  if (available.length === 0) {
    nextUsed = nextUsed.filter((id) => !themeWordIds.has(id));
    available = theme.words;
  }

  const entry = available[randomInt(available.length, rng)];
  const id = wordId(theme.id, entry.word);
  nextUsed.push(id);
  return { theme, entry, wordId: id, usedWordIds: nextUsed };
}

/** Monta uma rodada: palavra, impostores (com dicas diferentes) e quem começa. */
export function createRound(
  { players, impostorCount, themes, selectedThemeIds, hintsEnabled, usedWordIds },
  rng = Math.random,
) {
  const names = normalizePlayers(players);
  if (names.length < MIN_PLAYERS) throw new Error(`São necessários pelo menos ${MIN_PLAYERS} jogadores`);

  const { theme, entry, usedWordIds: nextUsed } = pickWord(themes, selectedThemeIds, usedWordIds, rng);
  const count = clampImpostors(impostorCount, names.length);
  const impostorOrder = shuffle(names.map((_, index) => index), rng).slice(0, count);

  const hints = {};
  if (hintsEnabled) {
    const shuffledHints = shuffle(entry.hints, rng);
    impostorOrder.forEach((playerIndex, i) => {
      hints[playerIndex] = shuffledHints[i % shuffledHints.length];
    });
  }

  return {
    round: {
      players: names,
      themeId: theme.id,
      themeName: theme.name,
      word: entry.word,
      impostors: [...impostorOrder].sort((a, b) => a - b),
      hints,
      starter: randomInt(names.length, rng),
    },
    usedWordIds: nextUsed,
  };
}
```

```js
// Leitura e escrita no localStorage. Falhas (modo privado, cota cheia, JSON inválido,
// armazenamento bloqueado) nunca quebram o jogo: a leitura devolve o padrão e a escrita é ignorada.

function storage() {
  const store = globalThis.localStorage;
  return store && typeof store.getItem === 'function' ? store : null;
}

export function loadJSON(key, fallback) {
  try {
    const raw = storage()?.getItem(key);
    return raw == null ? fallback : JSON.parse(raw);
  } catch {
    return fallback;
  }
}

  try {
    storage()?.setItem(key, JSON.stringify(value));
  } catch {
    // Sem armazenamento: o jogo continua funcionando, só não lembra das escolhas.
  }
}
```

```js
// Servidor estático mínimo para testar o jogo localmente (PC e celular na mesma rede).
// Só usa módulos nativos do Node. Responde 404 quando maiúsculas e minúsculas não batem,
// como o GitHub Pages, para pegar no Windows erros que só apareceriam depois do deploy.

import { createServer } from 'node:http';
import { realpathSync } from 'node:fs';
import { readFile, realpath, stat } from 'node:fs/promises';
import { extname, join, resolve, sep } from 'node:path';

const MIME_TYPES = {
  '.html': 'text/html; charset=utf-8',
  '.css': 'text/css; charset=utf-8',
  '.js': 'text/javascript; charset=utf-8',
  '.mjs': 'text/javascript; charset=utf-8',
  '.json': 'application/json; charset=utf-8',
  '.md': 'text/markdown; charset=utf-8',
  '.txt': 'text/plain; charset=utf-8',
  '.svg': 'image/svg+xml',
  '.png': 'image/png',
  '.ico': 'image/x-icon',
  '.webmanifest': 'application/manifest+json',
};

export function createStaticServer(rootDir) {
  const root = realpathSync.native(resolve(rootDir));
  return createServer((request, response) => {
    handle(root, request, response).catch(() => {
      if (response.headersSent) response.destroy();
      else send(response, 500, 'Erro interno');
    });
  });
}

async function handle(root, request, response) {
  let pathname;
  try {
    pathname = decodeURIComponent(new URL(request.url, 'http://localhost').pathname);
  } catch {
    send(response, 400, 'Endereço inválido');
    return;
  }

  let filePath = resolve(root, `.${pathname}`);
  if (filePath !== root && !filePath.startsWith(root + sep)) {
    send(response, 403, 'Acesso negado');
    return;
  }

  try {
    if ((await stat(filePath)).isDirectory()) filePath = join(filePath, 'index.html');
    const info = await stat(filePath);
    // realpath devolve a grafia real do disco: se for diferente, as maiúsculas não batem.
    if (!info.isFile() || (await realpath(filePath)) !== filePath) throw new Error('não encontrado');
  } catch {
    send(response, 404, 'Arquivo não encontrado');
    return;
  }

  const body = await readFile(filePath);
  response.writeHead(200, {
    'Content-Type': MIME_TYPES[extname(filePath).toLowerCase()] ?? 'application/octet-stream',
    'Cache-Control': 'no-store',
  });
  response.end(request.method === 'HEAD' ? undefined : body);
}

function send(response, status, message) {
  response.writeHead(status, { 'Content-Type': 'text/plain; charset=utf-8', 'Cache-Control': 'no-store' });
  response.end(message);
}
```

```js
// `npm start`: sobe o jogo em http://localhost:8080 e mostra o endereço para abrir no celular.
import { networkInterfaces } from 'node:os';
import { fileURLToPath } from 'node:url';
import { createStaticServer } from './static-server.mjs';

const root = fileURLToPath(new URL('..', import.meta.url));
const port = Number(process.env.PORT) || 8080;
const server = createStaticServer(root);

server.on('error', (error) => {
  if (error.code === 'EADDRINUSE') {
    console.error(`\n  A porta ${port} já está em uso. Use outra: $env:PORT=8081; npm start\n`);
    process.exit(1);
  }
  throw error;
});

server.listen(port, '0.0.0.0', () => {
  console.log('\n  Impostor rodando!\n');
  console.log(`  No PC:       http://localhost:${port}`);
  for (const { name, address } of lanAddresses()) {
    console.log(`  No celular:  http://${address}:${port}   (${name})`);
  }
  console.log('\n  O celular precisa estar na mesma rede Wi-Fi. Ctrl+C para parar.\n');
});

function lanAddresses() {
  return Object.entries(networkInterfaces()).flatMap(([name, nets]) =>
    (nets ?? [])
      .filter((net) => net.family === 'IPv4' && !net.internal)
      .map((net) => ({ name, address: net.address })),
  );
}
```

```js
// ---------- Utilidades de tela ----------

let screenShownAt = 0;

function showScreen(name) {
  state.screen = name;
  for (const section of $$('[data-screen]')) section.hidden = section.dataset.screen !== name;
  document.activeElement?.blur?.();
  window.scrollTo(0, 0);
  screenShownAt = performance.now();
}

/** true logo depois de uma troca de tela: um toque duplo não pode pular etapas. */
function tooSoon() {
  return performance.now() - screenShownAt < TAP_GUARD_MS;
}

/** Texto gigante que sempre cabe: --len recebe o tamanho do maior trecho (ver style.css). */
function setBigText(element, text) {
  element.textContent = text;
  const longest = Math.max(1, ...text.split(/[\s-]+/).map((part) => part.length));
  element.style.setProperty('--len', String(longest));
}

/**
 * Abre um diálogo e chama onConfirm se a pessoa tocar no botão value="confirm".
 * Sem suporte a <dialog> (iOS < 15.4), usa window.confirm com o texto de data-fallback.
 */
function openDialog(dialog, onConfirm = () => {}) {
  if (typeof dialog.showModal !== 'function') {
    if (window.confirm(dialog.dataset.fallback)) onConfirm();
    return;
  }
  if (dialog.open) return;
  dialog.returnValue = '';
  dialog.addEventListener(
    'close',
    () => {
      if (dialog.returnValue === 'confirm') onConfirm();
    },
    { once: true },
```

```js
/** Abre o jogo num "celular" emulado e coleta erros do console. */
export async function openGame({ width = 390, height = 844, reducedMotion = 'no-preference' } = {}) {
  const browser = await chromium.launch({ executablePath: CHROME, headless: true });
  const context = await browser.newContext({
    viewport: { width, height },
    deviceScaleFactor: 2,
    isMobile: true,
    hasTouch: true,

  });
  const page = await context.newPage();

  page.on('pageerror', (error) => errors.push(error.message));
  page.on('console', (message) => {
    if (message.type() === 'error') errors.push(message.text());
  });
  await page.goto(BASE_URL);
  return { browser, context, page, errors };
}

}
```

```js
    selectedThemeIds: settings.themeIds,
    hintsEnabled: settings.hints,
    usedWordIds: state.usedWordIds,
  });
  cancelDraw();
  clearReveal();
  clearResult();
  state.round = round;
  state.usedWordIds = usedWordIds;
  state.current = 0;
  saveJSON(USED_WORDS_KEY, usedWordIds);
  for (const element of $$('[data-theme-name]')) element.textContent = round.themeName;
  showPass();
}

function showPass() {
  const { round, current } = state;
  for (const element of $$('[data-progress]')) element.textContent = `${current + 1} / ${round.players.length}`;
  setBigText($('#pass-name'), round.players[current]);
  showScreen('pass');
}

function showReveal() {
  const { round, current } = state;
  const isImpostor = round.impostors.includes(current);
  // Preenche e mostra no mesmo passo: nunca aparece o conteúdo do jogador anterior.
  if (isImpostor) {
    const hint = round.hints[current];
    $('#impostor-hint').textContent = hint ?? '';
    $('#impostor-hint-box').hidden = !hint;
  } else {
    setBigText($('#reveal-word'), round.word);
  }
  $('#reveal-crew').hidden = isImpostor;
  $('#reveal-impostor').hidden = !isImpostor;
  $('#hide-btn').textContent = current === round.players.length - 1 ? 'Esconder e continuar' : 'Esconder e passar';
  showScreen('reveal');
}

/** Apaga o segredo da tela de revelação. */
function clearReveal() {
  $('#reveal-crew').hidden = true;
  $('#reveal-impostor').hidden = true;
  $('#reveal-word').textContent = '';
  $('#impostor-hint').textContent = '';
}

function clearResult() {
  $('#result-impostors').replaceChildren();
  $('#result-word').textContent = '';
}

function hideAndPass() {
  clearReveal();
  if (state.current < state.round.players.length - 1) {
    state.current += 1;
    showPass();
  } else {
    showStarter();
  }
}

function cancelDraw() {
  clearTimeout(drawTimer);
  drawTimer = 0;
}

/** Sorteio animado de quem começa (o resultado já veio pronto em round.starter). */
function showStarter() {
  const { round } = state;
  const nameElement = $('#starter-name');
  const revealButton = $('#open-reveal-confirm');
  $('#starter-label').hidden = true;
  $('#starter-announcement').textContent = '';
  nameElement.classList.remove('is-final');
  revealButton.disabled = true;
  showScreen('starter');

  const finish = () => {
    if (state.round !== round) return;
    drawTimer = 0;
    const name = round.players[round.starter];
    setBigText(nameElement, name);
    nameElement.classList.remove('is-drawing');
    nameElement.classList.add('is-final');
    $('#starter-label').hidden = false;
    $('#starter-announcement').textContent = `${name} começa!`;
    revealButton.disabled = false;
  };

  if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
    drawTimer = setTimeout(finish, 50); // um instante depois, para o leitor de tela anunciar
    return;
  }

  nameElement.classList.add('is-drawing');
  let index = Math.floor(Math.random() * round.players.length);
  let delay = 60;
  let elapsed = 0;
  const step = () => {
    if (state.round !== round) return; // saíram ou começaram outra rodada
    if (elapsed >= 2400) {
      finish();
      return;
    }
    index = (index + 1) % round.players.length;
    setBigText(nameElement, round.players[index]);
    elapsed += delay;
    delay = Math.min(delay * 1.15, 380);
    drawTimer = setTimeout(step, delay);
  };
  step();
}

function showResult() {
  cancelDraw();
  const { round } = state;
  $('#result-impostor-label').textContent = round.impostors.length > 1 ? 'Os impostores eram' : 'O impostor era';
  $('#result-impostors').replaceChildren(
    ...round.impostors.map((index) => {
      const item = document.createElement('li');
      const name = document.createElement('strong');
      name.textContent = round.players[index];
      item.append(name);
      if (round.hints[index]) {
        const hint = document.createElement('span');
        hint.className = 'impostor-hint';
        hint.textContent = `dica: ${round.hints[index]}`;
        item.append(hint);
      }
      return item;
    }),
  );
  setBigText($('#result-word'), round.word);
  showScreen('result');
}

function enterRound() {
  if (!historyGuard) {
    history.pushState({ impostorRound: true }, ''); // sempre dentro de um toque (ativação do usuário)
    historyGuard = true;
  }
  startRound();
}

function leaveRound() {
  cancelDraw();
  clearReveal();
  clearResult();
  state.round = null;
  renderSetup();
  showScreen('setup'); // troca a tela antes do history.back(): o popstate dele é ignorado
  if (historyGuard) {
    historyGuard = false;
    history.back();
  }
}

function confirmExit() {
  openDialog($('#exit-dialog'), leaveRound);
}

// "Voltar" do celular/navegador durante a rodada: pergunta antes de sair.
window.addEventListener('popstate', () => {
  if (state.screen === 'setup' || !historyGuard) return;
  historyGuard = false;
  if (state.screen === 'result') {
    leaveRound(); // rodada já terminou: nada a perder
    return;
  }
  history.pushState({ impostorRound: true }, '');
  historyGuard = true;
  confirmExit();
});

// Celular bloqueado ou troca de app com a palavra aberta: esconde e volta para "Passe o celular".
document.addEventListener('visibilitychange', () => {
  if (document.visibilityState === 'hidden' && state.screen === 'reveal') {
    clearReveal();
    showPass();
  }
});

$('#start-game').addEventListener('click', () => {
  if (tooSoon() || setupErrors().length > 0) return;
  enterRound();
});

$('#reveal-btn').addEventListener('click', () => {
  if (!tooSoon()) showReveal();
});

$('#hide-btn').addEventListener('click', () => {
  if (!tooSoon()) hideAndPass();
});

$('#open-reveal-confirm').addEventListener('click', () => {
  if (!tooSoon()) openDialog($('#reveal-dialog'), showResult);
});

$('#new-round').addEventListener('click', () => {
  if (!tooSoon()) startRound();
});

$('#back-to-setup').addEventListener('click', leaveRound);

for (const button of $$('[data-exit]')) button.addEventListener('click', confirmExit);
```

````markdown
# Impostor

Jogo de festa para jogar com os amigos usando **um celular só**. Todo mundo recebe a mesma palavra secreta, menos o **impostor**, que só recebe uma dica. Cada um fala uma palavra, e o grupo tenta descobrir quem está blefando.

## Como jogar

1. Escolha os jogadores, os temas e quantos impostores.
2. Passe o celular: cada um toca em **Revelar** e vê a palavra em segredo.
3. O app sorteia quem começa. Cada um fala **1 palavra** ligada à palavra secreta, em sentido horário.
4. Depois de 1 ou 2 voltas, discutam e votem. Toque em **Revelar impostor** para ver se acertaram.

## Rodar no computador

Precisa do [Node.js](https://nodejs.org) 20 ou mais novo. Não há dependências para instalar.

```bash
npm start
```

Abra o endereço "No PC" que aparece no terminal: `http://localhost:8080`.

## Jogar no celular (mesma rede Wi-Fi)

1. Rode `npm start` no computador.
2. No celular, conectado à **mesma rede Wi-Fi**, abra o endereço "No celular" que aparece no terminal (ex.: `http://192.168.0.10:8080`).
3. Se o celular não conseguir abrir, a rede do Windows provavelmente está como **Pública**. Em **Configurações → Rede e Internet → Wi-Fi → (sua rede)**, mude o **Tipo de perfil de rede** para **Privada**. Na primeira execução, permita o Node.js no Firewall do Windows.

Para usar outra porta (PowerShell): `$env:PORT=3000; npm start`

## Testes

```bash
npm test
```

Os testes cobrem as regras do jogo, o banco de palavras, o armazenamento e o servidor local.

## Editar as palavras

Os temas ficam em `js/words.js`, uma palavra por linha:

```
Pizza: Queijo, Forno, Fatia
```

Cada palavra tem **3 dicas** para o impostor. Depois de editar, rode `npm test`: ele avisa se alguma dica entrega a palavra, se há palavras repetidas ou se uma palavra é longa demais para a tela.

## Estrutura

| Caminho | O que é |
|---|---|
| `index.html`, `css/style.css` | Página única com todas as telas |
| `js/app.js` | Telas e interação |
| `js/game.js` | Regras do jogo (sorteios, limites, configurações) |
| `js/words.js` | Banco de palavras e dicas |
| `js/storage.js` | Salva as configurações no aparelho |
| `scripts/` | Servidor local (`npm start`) |
| `tests/` | Testes automáticos (`npm test`) |
````

```js
import { mkdirSync } from 'node:fs';
import { dirname, join } from 'node:path';
import { fileURLToPath } from 'node:url';
import { openGame, check, tapAfterGuard } from './lib.mjs';

const SIZES = [
  { width: 360, height: 640 },
  { width: 390, height: 844 },
  { width: 768, height: 1024 },
  { width: 1280, height: 800 },
];
const OUT = join(dirname(fileURLToPath(import.meta.url)), 'shots');
mkdirSync(OUT, { recursive: true });

async function noHorizontalScroll(page, label) {
  const overflow = await page.evaluate(() => document.documentElement.scrollWidth - window.innerWidth);
  check(overflow <= 0, `${label}: sem rolagem horizontal`);
}

/** Coloca um texto no elemento (como o setBigText faz) e confere que nada fica cortado. */
async function fits(page, selector, value, label) {
  const ok = await page.evaluate(
    ([sel, text]) => {
      const element = document.querySelector(sel);
      element.textContent = text;
      element.style.setProperty('--len', String(Math.max(...text.split(/[\s-]+/).map((part) => part.length))));
      return element.scrollWidth <= element.clientWidth + 1;
    },
    [selector, value],
  );
  check(ok, `${label}: "${value}" cabe sem cortar`);
}

for (const size of SIZES) {
  const tag = `${size.width}x${size.height}`;
  console.log(`\n# ${tag}`);
  const { browser, page, errors } = await openGame(size);
  const shot = (name) => page.screenshot({ path: join(OUT, `${tag}-${name}.png`) });
  try {
    await shot('1-configuracao');
    await noHorizontalScroll(page, 'configuração');
    await tapAfterGuard(page, '#start-game');
    await shot('2-passe');
    await fits(page, '#pass-name', 'Ana Carolina Souza', 'nome');
    let crewDone = false;
    let impostorDone = false;
    const total = await page.locator('.player-input').count();
    for (let i = 0; i < total; i += 1) {
      await tapAfterGuard(page, '#reveal-btn');
      const isImpostor = await page.locator('#reveal-impostor').isVisible();
      if (isImpostor && !impostorDone) {
        await shot('3-impostor');
        impostorDone = true;
      }
      if (!isImpostor && !crewDone) {
        await shot('3-palavra');
        await fits(page, '#reveal-word', 'Supermercado', 'palavra');
        await fits(page, '#reveal-word', 'Guaraná Antarctica', 'palavra');
        await fits(page, '#reveal-word', 'Parque de diversões', 'palavra');
        crewDone = true;
      }
      await noHorizontalScroll(page, 'revelação');
      await tapAfterGuard(page, '#hide-btn');
    }
    await page.locator('#starter-label').waitFor({ state: 'visible', timeout: 5000 });
    await shot('4-quem-comeca');
    await tapAfterGuard(page, '#open-reveal-confirm');
    await shot('5-confirmacao');
    await page.tap('#reveal-dialog button[value="confirm"]');
    await shot('6-resultado');
    await noHorizontalScroll(page, 'resultado');
    check(errors.length === 0, `sem erros no console ${errors.join(' | ')}`);
  } finally {
    await browser.close();
  }
}
console.log(`\nCapturas em ${OUT}`);
```
